function renderSpiral(maxN) { svg.innerHTML = ''; if (maxN === 0) { svg.innerHTML = `P_0 = 0`; return; } // 1. Gather Pell numbers dynamically let pells = [0]; for(let i = 1; i <= maxN + 1; i++) { pells.push(Number(computeGaloisCoefficients(i).y)); } // 2. Setup structural coordinates // We start with a core square of size P_1 x P_1 (1x1) let cx = 250, cy = 250; let currentDir = 0; // 0: Right, 1: Down, 2: Left, 3: Up // Scale dynamically to fit the 500x500 box based on the largest element const maxVal = pells[maxN + 1]; const scale = maxVal > 0 ? 350 / maxVal : 1; let pathData = `M ${cx} ${cy}`; // Render the initial P_1 square (1x1) let p1 = pells[1] * scale; drawSquare(cx, cy - p1, p1, `hsl(0, 65%, 55%)`, 0.1); // 3. Iterate through steps, accounting for the 2x multiplier for (let i = 2; i <= maxN; i++) { let p_prev = pells[i-1] * scale; let p_curr = pells[i] * scale; let strokeColor = `hsl(${(i * 45) % 360}, 65%, 45%)`; // To match 2*P_{n-1} + P_{n-2}, we draw two adjacent blocks // or shift our tracking framework across the dual-square expansion if (currentDir === 0) { // Expanding Right: attach a block of width (2 * p_curr) drawSquare(cx + p_prev, cy - p_curr, p_curr, strokeColor, 0.15); drawSquare(cx + p_prev + p_curr, cy - p_curr, p_curr, strokeColor, 0.15); // Draw matching smooth arc across the double block pathData += ` A ${p_curr} ${p_curr} 0 0 1 ${cx + p_prev + 2*p_curr} ${cy}`; cx = cx + p_prev + 2*p_curr; } else if (currentDir === 1) { // Expanding Down drawSquare(cx - p_curr, cy + p_prev, p_curr, strokeColor, 0.15); drawSquare(cx - p_curr, cy + p_prev + p_curr, p_curr, strokeColor, 0.15); pathData += ` A ${p_curr} ${p_curr} 0 0 1 ${cx} ${cy + p_prev + 2*p_curr}`; cy = cy + p_prev + 2*p_curr; } else if (currentDir === 2) { // Expanding Left drawSquare(cx - p_prev - p_curr, cy, p_curr, strokeColor, 0.15); drawSquare(cx - p_prev - 2*p_curr, cy, p_curr, strokeColor, 0.15); pathData += ` A ${p_curr} ${p_curr} 0 0 1 ${cx - p_prev - 2*p_curr} ${cy}`; cx = cx - p_prev - 2*p_curr; } else if (currentDir === 3) { // Expanding Up drawSquare(cx, cy - p_prev - p_curr, p_curr, strokeColor, 0.15); drawSquare(cx, cy - p_prev - 2*p_curr, p_curr, strokeColor, 0.15); pathData += ` A ${p_curr} ${p_curr} 0 0 1 ${cx} ${cy - p_prev - 2*p_curr}`; cy = cy - p_prev - 2*p_curr; } currentDir = (currentDir + 1) % 4; } // Overlay the valid mathematical spiral line let spiralPath = document.createElementNS("http://www.w3.org/2000/svg", "path"); spiralPath.setAttribute("d", pathData); spiralPath.setAttribute("fill", "none"); spiralPath.setAttribute("stroke", "#2c3e50"); spiralPath.setAttribute("stroke-width", "3"); svg.appendChild(spiralPath); } // Helper function to handle SVG element injections cleanly function drawSquare(x, y, size, color, opacity) { let rect = document.createElementNS("http://www.w3.org/2000/svg", "rect"); rect.setAttribute("x", x); rect.setAttribute("y", y); rect.setAttribute("width", size); rect.setAttribute("height", size); rect.setAttribute("fill", color); rect.setAttribute("fill-opacity", opacity); rect.setAttribute("stroke", color); rect.setAttribute("stroke-width", "1.5"); svg.appendChild(rect); }
